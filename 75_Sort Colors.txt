75. Sort Colors

class Solution:
    def sortColors(self, nums: List[int]) -> None:
        start,mid=0,0
        end=len(nums)-1
        while (mid<=end):
            if nums[mid]==0:
                nums[start],nums[mid]=nums[mid],nums[start]
                mid+=1
                start+=1
            elif nums[mid]==1:
                mid+=1
            elif nums[mid]==2:
                nums[mid],nums[end]=nums[end],nums[mid]
                end-=1
